### Position Calculations
```python
def calculate_distance(pos1, pos2):
    """
    Calculate Euclidean distance between two positions.
    
    Args:
        pos1: First position as (x, y) tuple
        pos2: Second position as (x, y) tuple
        
    Returns:
        Distance between positions
    """
    return np.sqrt((pos1[0] - pos2[0])**2 + (pos1[1] - pos2[1])**2)


def calculate_angle_to_target(current_pos, target_pos):
    """
    Calculate angle (in degrees) from current position to target position.
    
    Args:
        current_pos: Current position as (x, y, theta) tuple where theta is in radians
        target_pos: Target position as (x, y) tuple
        
    Returns:
        Angle to target in degrees
    """
    current_x, current_y, current_heading = current_pos
    target_x, target_y = target_pos
    
    # Calculate angle to target
    dx = target_x - current_x
    dy = target_y - current_y
    
    # Calculate angle in radians
    target_angle = np.arctan2(dy, dx)
    
    # Convert to degrees
    target_angle_deg = np.rad2deg(target_angle)
    current_heading_deg = np.rad2deg(current_heading)
    
    # Calculate angle difference
    angle_diff = (target_angle_deg - current_heading_deg) % 360
    if angle_diff > 180:
        angle_diff -= 360
    
    return angle_diff


def calculate_movement_vector(current_pos, target_pos, max_distance=0.5):
    """
    Calculate movement vector from current position to target position.
    
    Args:
        current_pos: Current position as (x, y) tuple
        target_pos: Target position as (x, y) tuple
        max_distance: Maximum distance to move in one step
        
    Returns:
        Movement vector as (dx, dy) tuple
    """
    current_x, current_y = current_pos
    target_x, target_y = target_pos
    
    # Calculate distance and direction
    dx = target_x - current_x
    dy = target_y - current_y
    distance = np.sqrt(dx**2 + dy**2)
    
    # Cap movement at max_distance
    if distance > max_distance:
        ratio = max_distance / distance
        dx *= ratio
        dy *= ratio
    
    return (dx, dy)
```

### Robot Movement
```python
def move_to_position(robot_controller, target_pos, speed=0.3):
    """
    Move the robot to the specified position.
    
    Args:
        robot_controller: Robot controller instance
        target_pos: Target position as (x, y) tuple
        speed: Movement speed
        
    Returns:
        True if movement was successful
    """
    current_x, current_y, current_heading = robot_controller.get_position()
    target_x, target_y = target_pos
    
    # Calculate angle to target
    angle_diff = calculate_angle_to_target((current_x, current_y, current_heading), target_pos)
    
    # Rotate to face the target
    if abs(angle_diff) > 5:
        robot_controller.ep_robot.chassis.move(
            x=0, y=0, z=angle_diff, xy_speed=speed
        ).wait_for_completed()
    
    # Calculate distance to target
    distance = calculate_distance((current_x, current_y), target_pos)
    
    # Move toward the target
    if distance > 0.1:
        robot_controller.ep_robot.chassis.move(
            x=distance, y=0, z=0, xy_speed=speed
        ).wait_for_completed()
    
    return True


def rotate_in_place(robot_controller, angle_deg, speed=0.3):
    """
    Rotate the robot by the specified angle.
    
    Args:
        robot_controller: Robot controller instance
        angle_deg: Angle to rotate in degrees
        speed: Rotation speed
        
    Returns:
        True if rotation was successful
    """
    robot_controller.ep_robot.chassis.move(
        x=0, y=0, z=angle_deg, xy_speed=speed
    ).wait_for_completed()
    
    return True


def drive_with_vector(robot_controller, x_vel, y_vel, z_vel):
    """
    Drive the robot with the specified velocity components.
    
    Args:
        robot_controller: Robot controller instance
        x_vel: Forward/backward velocity
        y_vel: Left/right velocity
        z_vel: Rotational velocity
        
    Returns:
        True if command was successful
    """
    robot_controller.ep_robot.chassis.drive_speed(
        x=x_vel, y=y_vel, z=z_vel
    )
    
    return True
```

### Vision Utilities
```python
def find_best_detection_by_confidence(detections, target_labels):
    """
    Find the detection with the highest confidence among the target labels.
    
    Args:
        detections: List of detection dictionaries
        target_labels: List of target label strings
        
    Returns:
        Best detection dictionary or None if no matching detection
    """
    best_detection = None
    max_confidence = 0.0
    
    for detection in detections:
        if detection["label"] in target_labels and detection["confidence"] > max_confidence:
            max_confidence = detection["confidence"]
            best_detection = detection
    
    return best_detection


def find_best_detection_by_center_distance(detections, target_labels, camera_center_x=320):
    """
    Find the detection closest to the camera center among the target labels.
    
    Args:
        detections: List of detection dictionaries
        target_labels: List of target label strings
        camera_center_x: X-coordinate of camera center
        
    Returns:
        Best detection dictionary or None if no matching detection
    """
    best_detection = None
    min_distance = float('inf')
    
    for detection in detections:
        if detection["label"] in target_labels:
            x1, _, x2, _ = detection["box"]
            box_center_x = (x1 + x2) / 2
            distance = abs(box_center_x - camera_center_x)
            
            if distance < min_distance:
                min_distance = distance
                best_detection = detection
    
    return best_detection


def get_detection_center(detection):
    """
    Get the center coordinates of a detection's bounding box.
    
    Args:
        detection: Detection dictionary
        
    Returns:
        (center_x, center_y) tuple
    """
    if detection is None:
        return None
    
    x1, y1, x2, y2 = detection["box"]
    center_x = (x1 + x2) / 2
    center_y = (y1 + y2) / 2
    
    return (center_x, center_y)


def get_detection_dimensions(detection):
    """
    Get the dimensions of a detection's bounding box.
    
    Args:
        detection: Detection dictionary
        
    Returns:
        (width, height) tuple
    """
    if detection is None:
        return None
    
    x1, y1, x2, y2 = detection["box"]
    width = x2 - x1
    height = y2 - y1
    
    return (width, height)
```
