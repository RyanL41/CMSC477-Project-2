# Lego Pickup Implementation

## Overview
The Lego pickup system will identify Lego blocks, center the robot, approach until the block is at an optimal position for pickup, and then use the robotic arm and gripper to grab the block.

## Implementation Details

### Lego Block Detection
```python
def detect_lego_blocks(object_detector, frame):
    """
    Detect Lego blocks in the frame.
    
    Args:
        object_detector: YOLO object detector instance
        frame: Camera frame
        
    Returns:
        Dictionary with detected blocks by label
    """
    if frame is None:
        return {}
    
    # Run YOLO detection
    detections, _ = object_detector.get_detections(frame)
    
    # Get detections for each lego block type
    detected_blocks = {
        "lego_small": object_detector.get_best_detection("lego_small", detections),
        "lego_med": object_detector.get_best_detection("lego_med", detections),
        "lego_big": object_detector.get_best_detection("lego_big", detections)
    }
    
    return detected_blocks
```

### Find Closest Block to Center
```python
def find_closest_block_to_center(detected_blocks):
    """
    Find the block closest to the center of the camera frame.
    
    Args:
        detected_blocks: Dictionary of detected blocks by label
        
    Returns:
        (detection, label) of the closest block, or (None, None) if no blocks detected
    """
    camera_center_x = 320  # Assuming camera width is 640px
    
    best_detection = None
    best_label = None
    min_distance = float('inf')
    
    for label, detection in detected_blocks.items():
        if detection is None:
            continue
        
        # Calculate distance to center
        x1, _, x2, _ = detection["box"]
        box_center_x = (x1 + x2) / 2
        distance = abs(box_center_x - camera_center_x)
        
        if distance < min_distance:
            min_distance = distance
            best_detection = detection
            best_label = label
    
    return best_detection, best_label
```

### Center Robot on Block
```python
def center_robot_on_block(robot_controller, detection):
    """
    Rotate the robot to center the detected block in the frame.
    
    Args:
        robot_controller: Robot controller instance
        detection: Block detection data
        
    Returns:
        True if the block is centered, False otherwise
    """
    if detection is None:
        return False
    
    x1, _, x2, _ = detection["box"]
    box_center_x = (x1 + x2) / 2
    camera_center_x = 320  # Assuming camera width is 640px
    
    # Calculate how far off-center the block is
    error_x = camera_center_x - box_center_x
    
    # If the error is small, the block is centered
    if abs(error_x) < 20:
        return True
    
    # Calculate rotation speed based on error
    z_vel = np.clip(-error_x * 0.05, -15, 15)
    
    # Rotate to center the block
    robot_controller.ep_robot.chassis.move(
        x=0, y=0, z=z_vel, xy_speed=0.3
    ).wait_for_completed()
    
    return False
```

### Approach Block
```python
def approach_block(robot_controller, detection):
    """
    Move the robot forward until the block is at the optimal pickup position.
    
    Args:
        robot_controller: Robot controller instance
        detection: Block detection data
        
    Returns:
        True if the block is at the optimal position for pickup, False otherwise
    """
    if detection is None:
        return False
    
    # Get the y-coordinate of the bottom of the bounding box
    _, y1, _, y2 = detection["box"]
    
    # Target y-position for pickup (y1 should be below this threshold)
    target_y = 230
    
    # If the block is already at the optimal position
    if y1 > target_y:
        # Stop the robot
        robot_controller.ep_robot.chassis.drive_speed(x=0, y=0, z=0)
        return True
    
    # Calculate forward speed based on distance from target position
    # Slow down as we get closer to the target
    distance_to_target = target_y - y1
    x_vel = min(0.1, max(0.05, distance_to_target * 0.001))
    
    # Move forward
    robot_controller.ep_robot.chassis.drive_speed(x=x_vel, y=0, z=0)
    
    return False
```

### Pickup Sequence
```python
def pickup_block(robot_controller):
    """
    Execute the sequence to pick up a block.
    
    Args:
        robot_controller: Robot controller instance
        
    Returns:
        True if pickup was successful
    """
    # Lower the arm
    robot_controller.ep_robot.robotic_arm.move(x=0, y=-100).wait_for_completed()
    
    # Close the gripper
    robot_controller.ep_robot.gripper.close(power=70)
    time.sleep(1.5)
    robot_controller.ep_robot.gripper.pause()
    
    # Raise the arm
    robot_controller.ep_robot.robotic_arm.move(x=0, y=70).wait_for_completed()
    
    return True
```

### Obstacle Avoidance During Pickup
```python
def check_for_obstacles_during_pickup(robot_controller, object_detector, apriltag_detector):
    """
    Check for obstacles during the pickup process and avoid them if necessary.
    
    Args:
        robot_controller: Robot controller instance
        object_detector: YOLO object detector instance
        apriltag_detector: AprilTag detector instance
        
    Returns:
        True if obstacles were detected and avoided, False otherwise
    """
    # Get current frame
    frame = robot_controller.get_frame()
    if frame is None:
        return False
    
    # Run YOLO detection to find other robots
    detections, _ = object_detector.get_detections(frame)
    robot_detection = object_detector.get_best_detection("robot", detections)
    
    # Convert frame to grayscale for AprilTag detection
    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    
    # Detect AprilTags
    apriltag_detections = apriltag_detector.find_tags(gray)
    
    # Check if we need to avoid any obstacles
    if robot_detection or apriltag_detections:
        # Handle robot obstacle
        if robot_detection:
            x1, _, x2, _ = robot_detection["box"]
            box_center_x = (x1 + x2) / 2
            camera_center_x = 320  # Assuming camera width is 640px
            
            if box_center_x < camera_center_x:
                # Robot is on the left, move right
                robot_controller.ep_robot.chassis.move(
                    x=0, y=-0.3, z=0, xy_speed=0.3
                ).wait_for_completed()
            else:
                # Robot is on the right, move left
                robot_controller.ep_robot.chassis.move(
                    x=0, y=0.3, z=0, xy_speed=0.3
                ).wait_for_completed()
            
            return True
        
        # Handle AprilTag obstacle (if no robot was detected)
        if apriltag_detections:
            # For simplicity, just handle the first AprilTag
            detection = apriltag_detections[0]
            
            # Get tag position relative to camera
            tag_relative_pos = apriltag_detector.get_tag_world_position(detection)
            
            # If the tag is close enough (< 1 box unit or ~0.26m)
            if np.linalg.norm(tag_relative_pos) < 1.0:
                # Determine if the tag is on the left or right
                if tag_relative_pos[0] < 0:
                    # Tag is on the left, move right
                    robot_controller.ep_robot.chassis.move(
                        x=0, y=-0.3, z=0, xy_speed=0.3
                    ).wait_for_completed()
                else:
                    # Tag is on the right, move left
                    robot_controller.ep_robot.chassis.move(
                        x=0, y=0.3, z=0, xy_speed=0.3
                    ).wait_for_completed()
                
                return True
    
    # No obstacle avoidance was performed
    return False
```

### Lego Pickup Loop
```python
def lego_pickup_loop(robot_controller, object_detector, apriltag_detector):
    """
    Main lego pickup loop.
    
    Args:
        robot_controller: Robot controller instance
        object_detector: YOLO object detector instance
        apriltag_detector: AprilTag detector instance
        
    Returns:
        True if pickup was successful, False otherwise
    """
    # First, check for obstacles
    obstacle_avoided = check_for_obstacles_during_pickup(robot_controller, object_detector, apriltag_detector)
    
    # If we had to avoid an obstacle, skip the rest of the loop
    if obstacle_avoided:
        return False
    
    # Get current frame
    frame = robot_controller.get_frame()
    if frame is None:
        return False
    
    # Detect lego blocks
    detected_blocks = detect_lego_blocks(object_detector, frame)
    
    # Find the closest block to the center
    best_detection, best_label = find_closest_block_to_center(detected_blocks)
    
    # If no blocks were detected, return False
    if best_detection is None:
        return False
    
    # Center the robot on the block
    is_centered = center_robot_on_block(robot_controller, best_detection)
    
    # If the robot is centered on the block, approach it
    if is_centered:
        # Get the latest frame and detection
        frame = robot_controller.get_frame()
        if frame is None:
            return False
        
        detected_blocks = detect_lego_blocks(object_detector, frame)
        best_detection = detected_blocks.get(best_label)
        
        if best_detection is None:
            return False
        
        # Approach the block
        is_at_pickup_position = approach_block(robot_controller, best_detection)
        
        # If the block is at the optimal position, pick it up
        if is_at_pickup_position:
            # Stop the robot
            robot_controller.ep_robot.chassis.drive_speed(x=0, y=0, z=0)
            
            # Pick up the block
            pickup_success = pickup_block(robot_controller)
            
            return pickup_success
    
    return False
```

## Integration with State Machine
The lego pickup system will be integrated into the state machine as the "Lego Pickup" state. Once the robot has successfully picked up a block, the state machine will transition to the "Path Planning (2: center line pre-waypoint)" state.
