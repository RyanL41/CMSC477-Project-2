# Lego Dropoff Implementation

## Overview
The Lego dropoff system will handle moving to the "closet", centering on it, approaching until the closet is out of view (indicating the robot is inside), and then releasing the held block.

## Implementation Details

### Detect Closet
```python
def detect_closet(object_detector, frame):
    """
    Detect the closet in the frame.
    
    Args:
        object_detector: YOLO object detector instance
        frame: Camera frame
        
    Returns:
        Detection data for the closet, or None if not detected
    """
    if frame is None:
        return None
    
    # Run YOLO detection
    detections, _ = object_detector.get_detections(frame)
    
    # Get detection for the closet
    closet_detection = object_detector.get_best_detection("closet", detections)
    
    return closet_detection
```

### Center Robot on Closet
```python
def center_robot_on_closet(robot_controller, detection):
    """
    Rotate the robot to center the detected closet in the frame.
    
    Args:
        robot_controller: Robot controller instance
        detection: Closet detection data
        
    Returns:
        True if the closet is centered, False otherwise
    """
    if detection is None:
        return False
    
    x1, _, x2, _ = detection["box"]
    box_center_x = (x1 + x2) / 2
    camera_center_x = 320  # Assuming camera width is 640px
    
    # Calculate how far off-center the closet is
    error_x = camera_center_x - box_center_x
    
    # If the error is small, the closet is centered
    if abs(error_x) < 20:
        return True
    
    # Calculate rotation speed based on error
    z_vel = np.clip(-error_x * 0.05, -15, 15)
    
    # Rotate to center the closet
    robot_controller.ep_robot.chassis.move(
        x=0, y=0, z=z_vel, xy_speed=0.3
    ).wait_for_completed()
    
    return False
```

### Approach Closet
```python
def approach_closet(robot_controller, object_detector):
    """
    Move the robot forward until the closet is no longer visible (meaning we're inside).
    
    Args:
        robot_controller: Robot controller instance
        object_detector: YOLO object detector instance
        
    Returns:
        True if the closet is no longer visible (we're inside), False otherwise
    """
    # Get current frame
    frame = robot_controller.get_frame()
    if frame is None:
        return False
    
    # Detect the closet
    closet_detection = detect_closet(object_detector, frame)
    
    # If the closet is not visible, we're likely inside it
    if closet_detection is None:
        # Stop the robot
        robot_controller.ep_robot.chassis.drive_speed(x=0, y=0, z=0)
        return True
    
    # Calculate the size of the closet bounding box
    x1, y1, x2, y2 = closet_detection["box"]
    box_width = x2 - x1
    box_height = y2 - y1
    
    # If the closet is taking up most of the frame, we're very close
    if box_width > 500 or box_height > 350:
        # Move forward slowly
        robot_controller.ep_robot.chassis.drive_speed(x=0.05, y=0, z=0)
    else:
        # Move forward at normal speed
        robot_controller.ep_robot.chassis.drive_speed(x=0.1, y=0, z=0)
    
    return False
```

### Dropoff Sequence
```python
def dropoff_block(robot_controller):
    """
    Execute the sequence to drop off a block.
    
    Args:
        robot_controller: Robot controller instance
        
    Returns:
        True if dropoff was successful
    """
    # Lower the arm
    robot_controller.ep_robot.robotic_arm.move(x=0, y=-100).wait_for_completed()
    
    # Open the gripper
    robot_controller.ep_robot.gripper.open(power=70)
    time.sleep(1.5)
    robot_controller.ep_robot.gripper.pause()
    
    # Raise the arm
    robot_controller.ep_robot.robotic_arm.move(x=0, y=70).wait_for_completed()
    
    # Back up slightly
    robot_controller.ep_robot.chassis.move(
        x=-0.3, y=0, z=0, xy_speed=0.3
    ).wait_for_completed()
    
    return True
```

### Lego Dropoff Loop
```python
def lego_dropoff_loop(robot_controller, object_detector):
    """
    Main lego dropoff loop.
    
    Args:
        robot_controller: Robot controller instance
        object_detector: YOLO object detector instance
        
    Returns:
        True if dropoff was successful, False otherwise
    """
    # Get current frame
    frame = robot_controller.get_frame()
    if frame is None:
        return False
    
    # Detect the closet
    closet_detection = detect_closet(object_detector, frame)
    
    # If the closet is not visible, we might be already inside or need to search for it
    if closet_detection is None:
        # Check if we're already inside the closet by trying to move forward a bit
        # and checking if the closet becomes visible
        robot_controller.ep_robot.chassis.move(
            x=0.1, y=0, z=0, xy_speed=0.3
        ).wait_for_completed()
        
        frame = robot_controller.get_frame()
        if frame is None:
            return False
        
        closet_detection = detect_closet(object_detector, frame)
        
        # If the closet is still not visible, we're likely inside
        if closet_detection is None:
            # Drop off the block
            dropoff_success = dropoff_block(robot_controller)
            return dropoff_success
        
        # If the closet became visible, we need to center on it again
        return False
    
    # Center the robot on the closet
    is_centered = center_robot_on_closet(robot_controller, closet_detection)
    
    # If the robot is centered on the closet, approach it
    if is_centered:
        # Approach the closet until we're inside (closet no longer visible)
        is_inside = approach_closet(robot_controller, object_detector)
        
        # If we're inside the closet, drop off the block
        if is_inside:
            dropoff_success = dropoff_block(robot_controller)
            return dropoff_success
    
    return False
```

## Integration with State Machine
The lego dropoff system will be integrated into the state machine as the "Lego Dropoff" state. Once the robot has successfully dropped off a block, the state machine will transition to the "Path Planning (4: center line post-dropoff)" state.
